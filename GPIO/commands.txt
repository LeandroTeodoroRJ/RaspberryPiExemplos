-- Install libgpiod library
sudo apt-get install libgpiod-dev
sudo apt-get install python3-libgpiod

-- List chips on board
gpiodetect

-- Set HIGH level GPIO number 4
sudo gpioset gpiochip0 4=1

-- Set LOW level GPIO number 4
sudo gpioset gpiochip0 4=0

-- Read GPIO line
sudo gpioget gpiochip0 1
	1
-- Copy service linux file
sudo cp gpio_scan.service /etc/systemd/system

-- SystemD service up
sudo systemctl enable gpio_scan.service
sudo systemctl start gpio_scan.service
sudo systemctl stop gpio_scan.service 
sudo systemctl disable gpio_scan.service
